\section{Quick Detection of a Wiener Process}

\begin{remark}
    Assume that we observe the trajectory of a Wiener process $X = (X_t)_{t \geq 0}$ whose drift may change from 0 to $\mu \neq 0$ at a random time $\theta$ (PS Ch. 22).
    
    Let $\PP(\theta = 0) = \pi$ and $\theta |_{\theta > 0} \sim \text{Exp}(\lambda)$ with $\lambda > 0$. We consider the space $(\Omega, \mc{F}, \PP_\pi, \pi \in [0,1])$ where
    \[ \PP_\pi = \pi \PP^0 + (1-\pi) \int_0^\infty \lambda e^{-\lambda s} \PP^s ds \]
    The process $X_t$ is such that
    \[ dX_t = \mu \mathbb{1}_{\{t \geq \theta\}} dt + \sigma dW_t, \quad X_0 = 0, \sigma > 0 \]
    which implies
    \[ X_t = \begin{cases} \sigma W_t & \text{if } \theta > t \\ \mu(t-\theta) + \sigma W_t & \text{if } t \geq \theta \end{cases} \]
    Here $\theta$ is unobservable. Our task is to find a stopping time $\tau^*$ of $X$ (relative to the natural filtration generated by $X_t$) that is as close as possible to $\theta$.
\end{remark}

\begin{remark}
    \vocab{Quickest detection problem}.
    The value function is defined as
    \[ V(\pi) = \inf_\tau \paren{ \underbrace{\PP_\pi(\tau < \theta)}_{\text{probability of a false alarm}} + c \underbrace{\EE_\pi [\tau - \theta]^+}_{\text{average delay in detecting the disorder}} } \]
    This can be reduced to
    \[ V(\pi) = \inf_\tau \EE_\pi \left[ 1 - \pi_\tau + c \int_0^\tau \pi_t dt \right], \]
    where $\pi_t = \PP_\pi(\theta \leq t \mid \mc{F}_t^X)$ for $t \geq 0$ is the \vocab{a posteriori probability}, with $\PP_\pi(\pi_0 = \pi) = 1$.
    
    By Bayes formula,
    \[ \pi_t = \pi \frac{d\PP^0}{d\PP_\pi}(t, X) + (1-\pi) \int_0^t \frac{d\PP^s}{d\PP_\pi}(t, X) \lambda e^{-\lambda s} ds \]
    where $\frac{d\PP^s}{d\PP_\pi}(t, X)$ is the \vocab{Radon-Nikodym density} of $\PP^s \mid \mc{F}_t^X$ with respect to $\PP_\pi \mid \mc{F}_t^X$.
    Furthermore,
    \[ 1 - \pi_t = (1-\pi) e^{-\lambda t} \frac{d\PP^t}{d\PP_\pi}(t, X) = (1-\pi) e^{-\lambda t} \frac{d\PP^\infty}{d\PP_\pi}(t, X) \]
    where $\PP^\infty$ corresponds to the law of $(\sigma W_t)_{t \geq 0}$.
\end{remark}

\begin{remark}
    The process for the likelihood ratio $\rho_t = \frac{\pi_t}{1-\pi_t}$ can be derived as
    \[ \rho_t = e^{\lambda t} z_t \paren{ \rho_0 + \lambda \int_0^t \frac{e^{-\lambda s}}{z_s} ds } = e^{Y_t} \paren{ \frac{\pi}{1-\pi} + \lambda \int_0^t e^{-Y_s} ds } \]
    where $z_t = \frac{d\PP^0}{d\PP^\infty}(t, X) \equiv \frac{d(\PP^0 \mid \mc{F}_t^X)}{d(\PP^\infty \mid \mc{F}_t^X)} = \exp \paren{ \frac{\mu}{\sigma^2} \paren{ X_t - \frac{\mu}{2} t } }$ and $Y_t = \lambda t + \frac{\mu}{\sigma^2} \paren{ X_t - \frac{\mu}{2} t }$.
    
    Applying Ito's formula to $z_t = f(X_t, t)$, we get
    \begin{align*}
        dz_t &= \frac{\mu}{\sigma^2} z_t dX_t - \left[ \frac{\mu^2}{2\sigma^2} z_t dt \right] + \frac{1}{2} \sigma^2 \frac{\mu^2}{\sigma^4} z_t dt \\
        &= \frac{\mu}{\sigma^2} z_t dX_t
    \end{align*}
    so that the dynamics of $\rho_t$ becomes
    \[ d\rho_t = \lambda(1 + \rho_t) dt + \frac{\mu}{\sigma^2} \rho_t dX_t \]
\end{remark}

\begin{remark}
    The problem is to compute
    \[ V(\pi) = \inf_{(\tau, d)} \EE_\pi \left[ \tau + a \mathbb{1}_{\{d=0, \theta=1\}} + b \mathbb{1}_{\{d=1, \theta=0\}} \right] \]
\end{remark}
